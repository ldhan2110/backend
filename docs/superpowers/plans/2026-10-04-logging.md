# Logging (Winston App + SQL) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Winston-based app logging (leveled, dev-pretty/prod-JSON) plus independent SqlMapper SQL debug logging with caller-service detection and timing.

**Architecture:** A `@Global` `LoggerModule` under `src/infra/logger/` registers `nest-winston` as the Nest system logger (knob: `LOG_LEVEL`) and provides a dedicated Winston logger for SQL (knob: `DB_LOGGING`, independent of `LOG_LEVEL`). `SqlMapper`'s two existing choke points (`query()` reads, `affected()` writes) log bound SQL + params + timing + calling service when `DB_LOGGING=true`.

**Tech Stack:** NestJS 11, `winston`, `nest-winston`, `@nestjs/config`, jest + ts-jest, pnpm.

## Global Constraints

- Package manager: **pnpm** (`pnpm add`, `pnpm test`).
- Tests: jest, files `*.spec.ts`, run via `pnpm test <pattern>`.
- Relative imports in new files: **extensionless** (match `src/infra/database/mapper/*`).
- Path aliases: `@infra/*` → `src/infra/*`, `@config/*` → `src/config/*`, `@common/*` → `src/common/*`.
- Prod detection: `process.env.NODE_ENV === 'production'` (lowercase, matches `.env.example`).
- Winston import style: `import * as winston from 'winston'`.
- No color in dev format (ANSI codes would corrupt the `[LEVEL]` token); plain brackets. Color is deferred (YAGNI).

```ts
// src/infra/logger/caller.ts

// ponytail: stack-trace heuristic, known ceiling — minified/deep-async stacks
// can blur frames → falls back to 'SqlMapper'. Only runs when DB_LOGGING=true.
export function callerFromStack(stack: string | undefined): string {
  const lines = stack?.split('\n') ?? [];
  for (const line of lines) {
    const m = line.match(/(\w+Service)\b/);
    if (m && m[1] !== 'SqlMapper') return m[1];
  }
  return 'SqlMapper';
}

export function callerService(): string {
  return callerFromStack(new Error().stack);
}
```

**Interfaces:**

- Produces:

  - `buildWinstonOptions(level: string): winston.LoggerOptions`

- [ ] **Step 1: Write `winston.config.ts`**

```ts
// src/infra/logger/winston.config.ts
import * as winston from 'winston';

const isProd = (): boolean => process.env.NODE_ENV === 'production';

export function buildFormat(): winston.Logform.Format {
  if (isProd()) {
    return winston.format.combine(winston.format.timestamp(), winston.format.json());
  }
  return winston.format.combine(
    winston.format.timestamp({ format: 'YYYY-MM-DD HH:mm:ss' }),
    winston.format.printf((info) => {
      const { timestamp, level, context, message } = info as {
        timestamp: string;
        level: string;
        context?: string;
        message: string;
      };
      return `[${timestamp}][${level.toUpperCase()}][${context ?? 'App'}]: ${message}`;
    }),
  );
}

export function buildWinstonOptions(level: string): winston.LoggerOptions {
  return {
    level,
    transports: [new winston.transports.Console({ format: buildFormat() })],
  };
}
```

```ts
// src/infra/logger/logger.module.ts
import { Global, Module } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { WinstonModule } from 'nest-winston';
import * as winston from 'winston';
import { buildFormat, buildWinstonOptions } from './winston.config';

export const DB_LOGGER = 'DB_LOGGER';

@Global()
@Module({
  imports: [
    WinstonModule.forRootAsync({
      inject: [ConfigService],
      useFactory: (config: ConfigService) =>
        buildWinstonOptions(config.get<string>('logging.level', 'info')),
    }),
  ],
  providers: [
    {
      provide: DB_LOGGER,
      useFactory: () =>
        winston.createLogger({
          level: 'debug',
          transports: [new winston.transports.Console({ format: buildFormat() })],
        }),
    },
  ],
  exports: [WinstonModule, DB_LOGGER],
})
export class LoggerModule {}
```

```ts
  private async query(
    text: string,
    params: Record<string, unknown>,
  ): Promise<Record<string, unknown>[]> {
    const { text: bound, values } = bind(text, params);
    if (!this.dbLog) return this.txHost.tx.query(bound, values);
    const caller = callerService();
    const t0 = performance.now();
    const rows = await this.txHost.tx.query(bound, values);
    const ms = (performance.now() - t0).toFixed(1);
    this.dbLogger.debug({
      context: caller,
      message: `${bound} -- params: ${JSON.stringify(values)} (${ms}ms)`,
    });
    return rows;
  }
```
